# PERHATIAN: Ganti path ini dengan lokasi folder dataset ko di Google Drive!
dataset_dir = '/content/drive/MyDrive/Clothes_Dataset' 

filepaths = []

# Panggil Base Model
base = MobileNetV2(weights='imagenet', include_top=False, input_shape=(224,224,3))
# Buka 50 layer terakhir biar akurasi meledak
base.trainable = True
for layer in base.layers[:-50]:
    layer.trainable = False
# Bangun Arsitektur
model = models.Sequential([
    base,
    layers.GlobalAveragePooling2D(),
    layers.BatchNormalization(), 
    layers.Dropout(0.4),         
    layers.Dense(5, activation='softmax') # FOKUS 5 KELAS
])
model.compile(optimizer=optimizers.Adam(learning_rate=1e-4), loss='categorical_crossentropy', metrics=['accuracy'])
# Callbacks untuk menyimpan otak terbaik
cb = [
    ModelCheckpoint('model_terbaik_5kelas.weights.h5', monitor='val_accuracy', save_best_only=True, save_weights_only=True, verbose=1),
    ReduceLROnPlateau(monitor='val_loss', factor=0.5, patience=3, verbose=1)
]
print("--- 🚀 MEMULAI TRAINING DI GPU T4 ---")
history = model.fit(
    train_generator, 
    epochs=50, 
    validation_data=val_generator,
    callbacks=cb,
    verbose=1
)

def get_gradcam(img, model):
    base_model = model.layers[0]
    extractor = tf.keras.models.Model(
        inputs=base_model.inputs, 
        outputs=[base_model.get_layer('out_relu').output, base_model.output]
    )
    
    with tf.GradientTape() as tape:
        conv_out, base_out = extractor(img)
        tape.watch(conv_out) 
        
        x = base_out
        for layer in model.layers[1:]:
            x = layer(x)
        
        preds = x
        loss = preds[:, tf.argmax(preds[0])]
        
    grads = tape.gradient(loss, conv_out)
    pooled_grads = tf.reduce_mean(grads, axis=(0, 1, 2))
    
    # Perkalian matriks yang aman (Keras 3 Compatible)
    hm = tf.reduce_sum(tf.multiply(conv_out[0], pooled_grads), axis=-1)
    hm = tf.maximum(hm, 0)
    hm = hm.numpy()
    
    max_val = np.max(hm)
    if max_val != 0:
        hm = hm / max_val
        
    return hm
